document.addEventListener('DOMContentLoaded', () => { let touchStartY = 0; // Suporte a toque (celulares e tablets) document.addEventListener('touchstart', (e) => { if (e.touches.length === 1) { touchStartY = e.touches.clientY; } }, { passive: true }); document.addEventListener('touchmove', (e) => { if (e.touches.length !== 1) return; const touchY = e.touches[0].clientY; const deltaY = touchStartY - touchY; const scrollTop = window.pageYOffset || document.documentElement.scrollTop; const scrollHeight = document.documentElement.scrollHeight; const clientHeight = document.documentElement.clientHeight; // Se chegou no final e puxou para baixo, volta para o topo if (scrollTop + clientHeight >= scrollHeight - 2 && deltaY > 0) { e.preventDefault(); window.scrollTo({ top: 0, behavior: 'smooth' }); } }, { passive: false }); // Suporte a mouse (computadores) document.addEventListener('wheel', (e) => { const scrollTop = window.pageYOffset || document.documentElement.scrollTop; const scrollHeight = document.documentElement.scrollHeight; const clientHeight = document.documentElement.clientHeight; // Se chegou no fim da página e rolou o mouse para baixo if (scrollTop + clientHeight >= scrollHeight - 2 && e.deltaY > 0) { e.preventDefault(); window.scrollTo({ top: 0, behavior: 'smooth' }); } }, { passive: false }); });
